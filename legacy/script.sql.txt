-- Importante: garantiza lectura del script como UTF-8
-- (evita problemas de acentos al importar en Windows)
SET NAMES utf8mb4;

DROP DATABASE IF EXISTS sistema_administrativo;
CREATE DATABASE sistema_administrativo
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE sistema_administrativo;

-- ------------------------------------------------------------
-- TABLA: usuarios (autenticación del sistema)
-- ------------------------------------------------------------
CREATE TABLE usuarios (
    id              INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre_usuario  VARCHAR(50)  NOT NULL UNIQUE,
    email           VARCHAR(100) NOT NULL UNIQUE,
    password_hash   VARCHAR(255) NOT NULL,
    nombre_completo VARCHAR(100) NOT NULL,
    rol             VARCHAR(20)  NOT NULL DEFAULT 'admin'
                    COMMENT 'admin | asistente',
    estado          TINYINT(1)   NOT NULL DEFAULT 1
                    COMMENT '1=activo, 0=inactivo',
    creado_en       TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- TABLA: estudiantes
-- ------------------------------------------------------------
CREATE TABLE estudiantes (
    id               INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    codigo           VARCHAR(20)  NOT NULL UNIQUE
                     COMMENT 'Código de estudiante',
    dni              CHAR(8)      NOT NULL UNIQUE,
    nombres          VARCHAR(80)  NOT NULL,
    apellidos        VARCHAR(80)  NOT NULL,
    email            VARCHAR(100) NULL,
    telefono         VARCHAR(20)  NULL,
    fecha_nacimiento DATE         NULL,
    direccion        VARCHAR(150) NULL,
    estado           TINYINT(1)   NOT NULL DEFAULT 1
                     COMMENT '1=activo, 0=inactivo',
    creado_en        TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_estudiantes_apellidos (apellidos)
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- TABLA: cursos
-- ------------------------------------------------------------
CREATE TABLE cursos (
    id         INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    codigo     VARCHAR(20)   NOT NULL UNIQUE,
    nombre     VARCHAR(100)  NOT NULL,
    descripcion TEXT         NULL,
    creditos   TINYINT UNSIGNED NOT NULL DEFAULT 3,
    horas      SMALLINT UNSIGNED NOT NULL DEFAULT 48,
    estado     TINYINT(1)    NOT NULL DEFAULT 1
               COMMENT '1=activo, 0=inactivo',
    creado_en  TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- TABLA: matriculas (relación N:M entre estudiantes y cursos)
-- ------------------------------------------------------------
CREATE TABLE matriculas (
    id             INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    estudiante_id  INT UNSIGNED NOT NULL,
    curso_id       INT UNSIGNED NOT NULL,
    periodo        VARCHAR(10)  NOT NULL
                   COMMENT 'Ej: 2026-02',
    fecha_matricula DATE        NOT NULL,
    estado         VARCHAR(15)  NOT NULL DEFAULT 'activa'
                   COMMENT 'activa | retirado',
    creado_en      TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_matricula_estudiante
        FOREIGN KEY (estudiante_id) REFERENCES estudiantes (id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_matricula_curso
        FOREIGN KEY (curso_id) REFERENCES cursos (id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    UNIQUE KEY uq_matricula (estudiante_id, curso_id, periodo),
    INDEX idx_matriculas_periodo (periodo)
) ENGINE=InnoDB;

-- ------------------------------------------------------------
-- VISTA: v_matriculas_detalle
-- Muestra las matrículas con datos del estudiante y del curso
-- mediante JOIN (requisito del trabajo: uso de JOIN y vista).
-- ------------------------------------------------------------
CREATE OR REPLACE VIEW v_matriculas_detalle AS
SELECT
    m.id                      AS matricula_id,
    m.periodo                 AS periodo,
    m.fecha_matricula         AS fecha_matricula,
    m.estado                  AS estado_matricula,
    e.id                      AS estudiante_id,
    e.codigo                  AS codigo_estudiante,
    e.dni                     AS dni,
    CONCAT(e.apellidos, ', ', e.nombres) AS estudiante,
    c.id                      AS curso_id,
    c.codigo                  AS codigo_curso,
    c.nombre                  AS curso,
    c.creditos                AS creditos
FROM matriculas m
INNER JOIN estudiantes e ON e.id = m.estudiante_id
INNER JOIN cursos c      ON c.id = m.curso_id;
